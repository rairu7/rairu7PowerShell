#■仕様
#ディレクトリ内を再帰的に検索し、ファイル名、行番号、列番号、行内容をtsv形式で出力する、ワンライナーPowerShell用プログラム。

#■使い方
#以下の4パラメーターを編集して、PowerShellにペースト。
# $directory
# $searchWord
# $outputFile
# $excludeExtensions
#必要に応じて除外ファイルを excludeExtensions に記載。

$directory = "検索対象ディレクトリ"; $searchWord = "TEL1"; $outputFile = "C:\Users\person\Documents\検索結果.txt"; if (Test-Path $outputFile) { Remove-Item $outputFile }; $excludeExtensions = @('jpg', 'mp4', 'gif', 'git'); Get-ChildItem -Path $directory -Recurse -File | ForEach-Object { $filePath = $_.FullName; $fileExtension = $_.Extension.TrimStart('.').ToLower(); if($excludeExtensions -contains $fileExtension) { return }; $lines = Get-Content $filePath; for ($lineNumber = 0; $lineNumber -lt $lines.Count; $lineNumber++) { $line = $lines[$lineNumber]; if ($line -match $searchWord){ $columnNumber = $line.IndexOf($searchWord) + 1; $line = $line -replace "`t", "    "; $output = "filePath't$fileExtension`t$($lineNumber + 1)`t$columnNumber't$line"; Add-Content -Path $outputFile -Value $output } } }
